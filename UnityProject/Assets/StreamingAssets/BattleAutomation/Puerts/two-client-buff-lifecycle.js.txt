/// two-client-buff-lifecycle 场景的 puerts controller
///
/// 场景目标：
/// - 两个客户端加入战斗
/// - 观察服务端权威 Buff 在两个玩家身上出现
/// - 继续观察该 Buff 倒计时结束后从两个玩家身上消失

let state = {
  joined: false,
  joinedFrame: 0,
  allPlayersPresentFrame: 0,
  observedTargetPlayerCount: false,
  observedExpectedBuffAppearance: false,
  observedExpectedBuffExpiry: false
};

let config = {
  minimumPlayerCount: 2,
  settleFrames: 30,
  expectedBuffId: 9001,
  buffApplyDelayFrames: 30,
  buffDurationFrames: 45
};

function countPlayersWithExpectedBuff(snapshot) {
  if (!snapshot.players) {
    return 0;
  }

  let count = 0;
  for (let i = 0; i < snapshot.players.length; i++) {
    const player = snapshot.players[i];
    if (!player || !player.activeBuffs) {
      continue;
    }

    for (let j = 0; j < player.activeBuffs.length; j++) {
      const buff = player.activeBuffs[j];
      if (buff && (buff.buffId | 0) === config.expectedBuffId) {
        count++;
        break;
      }
    }
  }

  return count;
}

module.exports = {
  initialize(configJson) {
    const parsed = JSON.parse(configJson || "{}");
    config.minimumPlayerCount = parsed.minimumPlayerCount || 2;
    config.settleFrames = parsed.settleFrames || 30;
    config.expectedBuffId = parsed.expectedBuffId || 9001;
    config.buffApplyDelayFrames = parsed.buffApplyDelayFrames || 30;
    config.buffDurationFrames = parsed.buffDurationFrames || 45;

    state = {
      joined: false,
      joinedFrame: 0,
      allPlayersPresentFrame: 0,
      observedTargetPlayerCount: false,
      observedExpectedBuffAppearance: false,
      observedExpectedBuffExpiry: false
    };
  },

  tryGetInput(payloadJson) {
    return JSON.stringify({ hasInput: false, dx: 0.0, dy: 0.0 });
  },

  evaluate(snapshotJson) {
    const snapshot = JSON.parse(snapshotJson || "{}");

    if (snapshot.joined && !state.joined) {
      state.joined = true;
      state.joinedFrame = snapshot.localFrame | 0;
    }

    if ((snapshot.activePlayerCount | 0) >= config.minimumPlayerCount) {
      state.observedTargetPlayerCount = true;
      if (!state.allPlayersPresentFrame) {
        state.allPlayersPresentFrame = snapshot.localFrame | 0;
      }
    }

    if (!state.joined) {
      return JSON.stringify({ completed: false, passed: false, reason: "" });
    }

    // 等待所有玩家到齐，最多等 300 帧（~10 秒）
    const joinElapsedFrames = (snapshot.localFrame | 0) - state.joinedFrame;
    if (!state.observedTargetPlayerCount) {
      if (joinElapsedFrames >= 900) {
        return JSON.stringify({
          completed: true,
          passed: false,
          reason: "timeout-waiting-for-players need=" + config.minimumPlayerCount + " have=" + (snapshot.activePlayerCount | 0)
        });
      }
      return JSON.stringify({ completed: false, passed: false, reason: "" });
    }

    // Buff 生命周期计时器从"所有玩家都在"时开始
    const elapsedFrames = (snapshot.localFrame | 0) - state.allPlayersPresentFrame;
    const playersWithExpectedBuff = countPlayersWithExpectedBuff(snapshot);

    if (state.observedTargetPlayerCount &&
        playersWithExpectedBuff >= config.minimumPlayerCount) {
      state.observedExpectedBuffAppearance = true;
    }

    if (state.observedExpectedBuffAppearance &&
        playersWithExpectedBuff === 0 &&
        elapsedFrames >= config.buffApplyDelayFrames + config.buffDurationFrames) {
      state.observedExpectedBuffExpiry = true;
    }

    const completionFrame =
      config.buffApplyDelayFrames + config.buffDurationFrames + config.settleFrames;

    if (elapsedFrames >= completionFrame) {
      if (state.observedExpectedBuffAppearance && state.observedExpectedBuffExpiry) {
        return JSON.stringify({
          completed: true,
          passed: true,
          reason: "buff-lifecycle-complete buffId=" + config.expectedBuffId
        });
      }

      if (!state.observedExpectedBuffAppearance) {
        return JSON.stringify({
          completed: true,
          passed: false,
          reason: "buff-never-appeared buffId=" + config.expectedBuffId
        });
      }

      return JSON.stringify({
        completed: true,
        passed: false,
        reason: "buff-never-expired buffId=" + config.expectedBuffId
      });
    }

    return JSON.stringify({ completed: false, passed: false, reason: "" });
  },

  dispose() {}
};
